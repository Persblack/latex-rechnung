\newcounter{itemnum}
\newcommand{\fmtnum}[1]{\ifnum#1<10 0\the#1\else\the#1\fi}

\FPset{\invoiceSubtotal}{0}
\newcommand{\allFeeRows}{}

% \Fee accumulates rows into \allFeeRows.
% Bug fix: each row is fully expanded into \curRow via \edef (embedding literal
% argument text), then the EXPANDED content of \curRow is appended to \allFeeRows
% via \expandafter. This prevents later \Fee calls from overwriting earlier rows,
% which happened when row tokens referenced shared macro variables like \curDesc.
\newcommand{\Fee}[3]{%
  \stepcounter{itemnum}%
  \FPmul{\myLineTot}{#2}{#3}%
  \FPround{\myLineTotR}{\myLineTot}{2}%
  \FPadd{\myNewSub}{\invoiceSubtotal}{\myLineTotR}%
  \FPround{\myNewSubR}{\myNewSub}{2}%
  \xdef\invoiceSubtotal{\myNewSubR}%
  \edef\curRow{%
    \noexpand\textcolor{brand}{\noexpand\textbf{\fmtnum{\value{itemnum}}}} &
    \noexpand\textbf{#1} &
    #3 &
    \noexpand\fmtmoney{#2} &
    \noexpand\fmtmoney{\myLineTotR} \noexpand\\[1pt]%
    \noexpand\arrayrulecolor{lightgray}\noexpand\hline
  }%
  \expandafter\gappto\expandafter\allFeeRows\expandafter{\curRow}%
}

% \FeeDiscount: per-line discount (Nachlass) sub-row, emitted right after a
% \Fee. The visible amount is the reduction; the invoice TOTAL comes from the
% Go VAT layer (\NetTotal), which already nets out discounts, so this row is
% display-only (we also keep \invoiceSubtotal consistent for completeness).
% #1 = percent number (e.g. "10") or empty for a fixed-amount discount,
% #2 = the discount amount in € (e.g. "9.00").
% Appended raw via \gappto (no \edef needed: #1/#2 are literal arguments, and
% \ifstrempty is evaluated when the table is typeset).
\newcommand{\FeeDiscount}[2]{%
  \FPsub{\myNewSub}{\invoiceSubtotal}{#2}%
  \FPround{\myNewSubR}{\myNewSub}{2}%
  \xdef\invoiceSubtotal{\myNewSubR}%
  \gappto\allFeeRows{%
    &
    \textcolor{midgray}{\itshape\ifEnglish Discount\else Nachlass\fi\ifstrempty{#1}{}{ #1\,\%}} &
    &
    &
    \textcolor{midgray}{-\fmtmoney{#2}} \\[1pt]%
    \arrayrulecolor{lightgray}\hline
  }%
}
