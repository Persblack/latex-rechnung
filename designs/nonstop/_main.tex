% =============================================================
%  BRIEFKOPF-OVERLAY (nur Seite 1)
%  Wird dem ersten Textbaustein vorangestellt; overlay-tikzpictures
%  haben Breite und Höhe 0 und stören den Umbruch deshalb nicht.
% =============================================================
% \atpage{<anchor>}{<x mm>}{<y mm von oben>}{<Inhalt>}
% Mehrzeilige Inhalte werden in \parbox[t] gesetzt: damit ist der
% base-Anker die Grundlinie der ERSTEN Zeile (ein nacktes mehrzeiliges
% tikz-Node verankert sonst auf der letzten Zeile).
\newcommand{\atpage}[4]{%
  \begin{tikzpicture}[remember picture, overlay]
    \node[anchor=#1, inner sep=0pt, outer sep=0pt]
      at ([xshift=#2mm, yshift=-#3mm]current page.north west) {#4};
  \end{tikzpicture}%
}

\newcommand{\unitsuffix}[1]{\ifstrempty{#1}{}{\,#1}}

% \FeeNote: optionale Zusatzzeile, vom Go-Layer in das
% Beschreibungsargument eingebettet. \protected\def ist notwendig, weil
% \Fee die Zeile per \edef aufbaut und \newline nicht expandierbar ist.
\protected\def\FeeNote#1{\newline{\notefont #1}}

% \LineDiscount wird vom gemeinsamen Template direkt VOR dem zugehörigen
% \Fee ausgegeben; die Werte werden zwischengespeichert und nach der
% Positionszeile als eigene Zeile gesetzt.
\newcommand{\LDpercent}{}
\newcommand{\LDamount}{}
\def\LDcents{0}
\newcommand{\LineDiscount}[3]{\gdef\LDpercent{#1}\gdef\LDcents{#2}\gdef\LDamount{#3}}

\newcommand{\Fee}[4]{%
  \stepcounter{itemnum}%
  \FPmul{\myLineTot}{#2}{#3}%
  \FPround{\myLineTotR}{\myLineTot}{2}%
  \FPadd{\myNewSub}{\invoiceSubtotal}{\myLineTotR}%
  \FPround{\myNewSubR}{\myNewSub}{2}%
  \xdef\invoiceSubtotal{\myNewSubR}%
  \edef\curRow{%
    \the\value{itemnum} &
    #1 &
    \noexpand\fmtqty{#3}\noexpand\unitsuffix{#4} &
    \noexpand\fmtmoney{#2}\noexpand\,\noexpand\texteuro &
    \noexpand\fmtmoney{\myLineTotR}\noexpand\,\noexpand\texteuro
    \noexpand\\[2.2mm]%
  }%
  \expandafter\gappto\expandafter\allFeeRows\expandafter{\curRow}%
  \ifnum\LDcents=0 \else
    \edef\LDcall{\noexpand\FeeDiscount{\LDpercent}{\LDamount}}\LDcall
    \gdef\LDpercent{}\gdef\LDcents{0}\gdef\LDamount{}%
  \fi
}

% \FeeDiscount: Nachlasszeile unter der Position. Angezeigt wird die
% Minderung; der Rechnungsbetrag kommt aus der Go-Schicht (\NetTotal),
% die den Nachlass bereits abgezogen hat.
\newcommand{\FeeDiscount}[2]{%
  \FPsub{\myNewSub}{\invoiceSubtotal}{#2}%
  \FPround{\myNewSubR}{\myNewSub}{2}%
  \xdef\invoiceSubtotal{\myNewSubR}%
  \gappto\allFeeRows{%
    & \textit{\ifEnglish Discount\else Nachlass\fi\ifstrempty{#1}{}{ #1\,\%}} & & &
    $-$\fmtmoney{#2}\,\texteuro \\[2.2mm]%
  }%
}
